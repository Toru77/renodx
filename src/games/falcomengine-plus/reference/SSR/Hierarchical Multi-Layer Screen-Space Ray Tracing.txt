vec3 reproject ( vec2 at , float depth ) {
const float fy = 1. f / tan_fovy , fx = fy / aspect , inv_fx = 1. f / fx ;
const vec2 inv_screendim = 1. f / vec2 ( screendim ) ;
vec2 clip = at * inv_screendim * 2. f - 1. f ;
float eye_z = - depth * cam_near_far . y ;
float eye_x = ( clip . x * - eye_z ) * inv_fx ;
float eye_y = ( clip . y * - eye_z ) * tan_fovy ;
return vec3 ( eye_x , eye_y , eye_z ) ;
}

void setupFor ( in Ray ray ) {
// clip near plane
float rayLength = ( - ray . orig . z - ray . dir . z * MAX_LEN ) <
cam_near_far . x ? ( cam_near_far . x + ray . orig . z ) / - ray . dir . z :
MAX_LEN ;
vec3 csEndPoint = ray . orig + ray . dir * rayLength ;
vec4 H0 = proj * vec4 ( ray . orig , 1.0) , H1 = proj * vec4 ( csEndPoint ,
1.0) ;
state . k0 = 1.0 / H0 . w ; float k1 = 1.0 / H1 . w ;
state . Q0 = ray . orig * state . k0 ; vec3 Q1 = csEndPoint * k1 ;
state . P0 = H0 . xy * state . k0 ; vec2 P1 = H1 . xy * k1 ;
state . P0 = (( state . P0 + vec2 (1) ) /2) * screendim ; P1 = (( P1 +
vec2 (1) ) /2) * screendim ;
// [ optional clipping ]
# ifdef OPTIONAL_CLIPPING
float xMax = screendim . x - 0.5 , xMin = 0.5 , yMax = screendim . y - 0.5 ,
yMin = 0.5;
float alpha = 0.0;
if (( P1 . y > yMax ) || ( P1 . y < yMin ) )
alpha = ( P1 . y - (( P1 . y > yMax ) ? yMax : yMin ) ) / ( P1 . y state . P0 . y ) ;
if (( P1 . x > xMax ) || ( P1 . x < xMin ) )
alpha = max ( alpha , ( P1 . x - (( P1 . x > xMax ) ? xMax : xMin ) ) / ( P1 . x state . P0 . x ) ) ;
P1 = (1 - alpha ) * P1 + alpha * state . P0 ;
k1 = (1 - alpha ) * k1 + alpha * state . k0 ;
Q1 = (1 - alpha ) * Q1 + alpha * state . Q0 ;
# endif
P1 += vec2 (( distanceSquared ( state . P0 , P1 ) < 0.0001) ? 0.01 : 0.0) ;
vec2 delta = P1 - state . P0 ;
state . permute = false ;
if ( abs ( delta . x ) < abs ( delta . y ) ) {
state . permute = true ;
delta = delta . yx ; state . P0 = state . P0 . yx ; P1 = P1 . yx ;
}
state . stepDir = sign ( delta . x ) ;
float invdx = state . stepDir / delta . x ;
state . dQ = ( Q1 - state . Q0 ) * invdx ;
state . dk = ( k1 - state . k0 ) * invdx ;
state . dP = vec2 ( state . stepDir , delta . y * invdx ) ;
state . dP *= stride ; state . dQ *= stride ; state . dk *= stride ;
state . end = P1 . x * state . stepDir ;
}

void setupFor ( in Ray ray ) {

vec4 H0 = proj * vec4 ( ray . orig , 1.0) ;
state . k0 = 1.0 / H0 . w ; state . Q0 = ray . orig * state . k0 ;
state . P0 = (( H0 . xy * state . k0 + 1. f ) /2. f ) * screendim ;
vec2 P1 ;
// compute distance to screen borders and clip in screen space
vec4 Hclip = proj * vec4 ( ray . orig + ray . dir , 1. f ) ;
vec2 Pclip = (( Hclip . xy / Hclip . w + 1. f ) /2. f ) * screendim ;
vec2 dir_screen = normalize ( Pclip - state . P0 ) ;
vec2 dest_screen = step (0 , dir_screen ) * ( screendim - ivec2 (1) ) ;
vec2 dist_screen = ( dest_screen - state . P0 ) / dir_screen ;
P1 = state . P0 + dir_screen * min ( dist_screen .x , dist_screen . y ) ;

vec3 csFar = reproject ( P1 , 1. f ) ;
vec3 csFarDir = normalize ( csFar ) ;
vec3 n = cross ( csFarDir , cross ( ray . dir , csFarDir ) ) ;
float len = dot ( csFar - ray . orig , n ) / dot ( ray . dir , n ) ;
vec3 ray_end = ray . orig + len * ray . dir ;
// check for error , clip far plane
if ( - ray_end . z > cam_near_far . y || len < 0)
ray_end = ray . orig + ray . dir * ( cam_near_far . y + ray . orig . z ) /
- ray . dir . z ;
vec4 H1 = proj * vec4 ( ray_end , 1.0) ; float k1 = 1. f / H1 . w ; vec3 Q1 =
ray_end * k1 ;
P1 = (( H1 . xy * k1 + 1. f ) /2. f ) * screendim ;
